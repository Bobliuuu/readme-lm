{%- assign d = site.data.l3 -%}{%- assign E = d.E -%}{%- assign A = d.A -%}{%- assign O = d.O -%}
{%- assign h = "0,0,0,0,0" | split: "," -%}
{%- assign toks = d.prompt -%}{%- assign gen = "" | split: "" -%}{%- assign cache = "" | split: "" -%}
{%- for step in (0..40) -%}
  {%- if step < toks.size -%}{%- assign t = toks[step] -%}{%- else -%}{%- assign t = gen | last -%}{%- endif -%}
  {%- assign nh = "" | split: "" -%}
  {%- for i in (0..4) -%}{%- assign acc = E[t][i] -%}{%- for j in (0..4) -%}{%- assign acc = h[j] | times: A[i][j] | plus: acc -%}{%- endfor -%}{%- assign acc = acc | at_least: 0 -%}{%- assign nh = nh | push: acc -%}{%- endfor -%}
  {%- assign h = nh -%}{%- assign cache = cache | push: nh -%}
  {%- assign gate = toks.size | minus: 1 -%}{%- if step >= gate -%}
    {%- assign best = -1000000 -%}{%- assign arg = 0 -%}
    {%- for v in (0..5) -%}{%- assign lg = 0 -%}{%- for j in (0..4) -%}{%- assign lg = h[j] | times: O[v][j] | plus: lg -%}{%- endfor -%}{%- if lg > best -%}{%- assign best = lg -%}{%- assign arg = v -%}{%- endif -%}{%- endfor -%}
    {%- assign gen = gen | push: arg -%}
    {%- if arg == d.eos -%}{%- break -%}{%- endif -%}
  {%- endif -%}
{%- endfor -%}
GEN {{ gen | join: " " }}
CACHE_LEN {{ cache | size }}
